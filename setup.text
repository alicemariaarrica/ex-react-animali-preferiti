    <script type="text/babel">
        const { useState } = React;
        const Counter = () => {

            const [count, setCount] = useState(0);

            const incrementa = () => setCount(curr => curr + 1);


            return (
                <main>
                    <h1>Applicazione React</h1>
                    <p>Count: {count}</p>
                    <button onClick={incrementa}>Incrementa</button>
                </main>
            )
        }
    </script>





    <script type="text/babel">
        const { createRoot } = ReactDOM;
        const elemento = document.querySelector(".elementoSpecifico");

        //console.log(elemento); non mi esce <div class="elementoSpecifico"></div> perchè tutto questo codice dovrebbe essere nel body per visualizzare il div ma vabbè

        const root = createRoot(elemento);

        //console.log(root);

        root.render(<Counter />)
    </script>
